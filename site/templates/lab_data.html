Quick start

import pandas as pd

BASE = "{{ base }}/v1/latest/bulk"
WEB = {"User-Agent": "data-notebook"}   # the host blocks Python's default user agent, so name your client
obs = pd.read_parquet(f"{BASE}/observations.parquet", storage_options=WEB)   # one row per country, indicator and year
countries = pd.read_parquet(f"{BASE}/entities.parquet", storage_options=WEB).query("type == 'country'")

# Latest life expectancy per country since 2018
le = (obs[(obs.indicator_id == "life-expectancy") & (obs.year >= 2018)]
      .sort_values("year").groupby("entity_id").tail(1))

# Wide table: one column per indicator (latest value since 2018)
latest = obs[obs.year >= 2018].sort_values("year").groupby(["entity_id", "indicator_id"]).value.last()
wide = latest.unstack()
print(wide[["gdp-per-capita-ppp", "life-expectancy"]].corr(method="spearman"))

INSTALL httpfs; LOAD httpfs;

CREATE VIEW obs AS
  SELECT * FROM '{{ base }}/v1/latest/bulk/observations.parquet';

-- Latest value per country for two indicators, side by side
SELECT entity_id,
       max_by(value, year) FILTER (WHERE indicator_id = 'gdp-per-capita-ppp') AS income,
       max_by(value, year) FILTER (WHERE indicator_id = 'life-expectancy')    AS life_expectancy
FROM obs
WHERE year >= 2018
GROUP BY entity_id
ORDER BY life_expectancy DESC;
